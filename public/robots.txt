# Digital Penta — crawler policy
# One wildcard group so every crawler (Googlebot, Bingbot, AI answer engines,
# social preview bots) gets the SAME rules. Named "Allow: /" groups were
# removed because a bot that matches a named group ignores the wildcard
# group's Disallow rules — which previously let named bots crawl /dashboard/
# and /auth/.

User-agent: *
Allow: /
Disallow: /api/
Disallow: /dashboard/
Disallow: /auth/
Disallow: /login
Disallow: /signup
Disallow: /forgot-password
Disallow: /reset-password
Disallow: /data-request
Disallow: /unsubscribe

Sitemap: https://digitalpenta.com/sitemap.xml
